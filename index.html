function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var datos = e.parameter || {}; var hojaDestino = datos.hoja; if (hojaDestino === 'ENTRADA') { var sheetEntrada = ss.getSheetByName('ENTRADA'); sheetEntrada.appendRow([ '✅️', datos.fecha || '', datos.hrEntrada || '', datos.hrSalida || '❓️', datos.nombre || '', datos.cedula || '❓️', datos.observaciones || '❓️' ]); } else if (hojaDestino === 'GUARDIA') { var sheetGuardia = ss.getSheetByName('GUARDIA'); sheetGuardia.appendRow([ '✅️', datos.dia || '', datos.nombreAfe2 || '', datos.entradaAfe2 || '', datos.nombreAfeOf || '', datos.salidaAfeOf || '' ]); } return ContentService .createTextOutput(JSON.stringify({ result: 'success' })) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = (e && e.parameter && e.parameter.hoja) ? e.parameter.hoja : 'ENTRADA'; var limite = Number((e && e.parameter && e.parameter.limite) ? e.parameter.limite : 8); var sheet = ss.getSheetByName(hoja); if (!sheet) { return ContentService .createTextOutput(JSON.stringify({ rows: [] })) .setMimeType(ContentService.MimeType.JSON); } var lastRow = sheet.getLastRow(); var lastCol = sheet.getLastColumn(); var rows = []; if (lastRow > 0 && lastCol > 0) { var startRow = Math.max(1, lastRow - limite + 1); var values = sheet.getRange(startRow, 1, lastRow - startRow + 1, lastCol).getDisplayValues(); rows = values.reverse(); } return ContentService .createTextOutput(JSON.stringify({ hoja: hoja, rows: rows })) .setMimeType(ContentService.MimeType.JSON); }
